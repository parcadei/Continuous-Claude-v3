## Escape Hatches

### `--external-db` (Bring Your Own Database)

Use an existing PostgreSQL instance:

```bash
rigg init --external-db "postgres://user:pass@host:5432/rigg"
```

Requirements:
- PostgreSQL 14+ with pgvector extension
- User with CREATE TABLE permissions
- rigg runs migrations automatically

```python
# rigg/cli/init.py

async def init_with_external_db(db_url: str):
    """Initialize with external PostgreSQL."""

    # Validate connection
    try:
        pool = await asyncpg.create_pool(db_url)
        await pool.execute("SELECT 1")
    except Exception as e:
        console.print(f"[red]Cannot connect to database: {e}[/red]")
        raise typer.Exit(1)

    # Check pgvector
    has_pgvector = await pool.fetchval("""
        SELECT EXISTS (
            SELECT 1 FROM pg_extension WHERE extname = 'vector'
        )
    """)

    if not has_pgvector:
        console.print("[yellow]Installing pgvector extension...[/yellow]")
        await pool.execute("CREATE EXTENSION IF NOT EXISTS vector")

    # Run migrations
    await run_migrations(pool)

    # Save config
    config = load_config()
    config['database'] = {'url': db_url, 'external': True}
    save_config(config)

    # Skip PostgreSQL container in docker-compose
    config['docker']['skip_services'] = ['postgres']
    save_config(config)
```

### `--no-docker` (SQLite Lite Mode)

For environments without Docker (CI, minimal setups):

```bash
rigg init --no-docker
```

Limitations:
- SQLite instead of PostgreSQL (no pgvector, no embeddings)
- In-memory cache instead of Redis (no cross-process blackboard)
- No sandbox isolation (code runs in main process)
- Single-agent only (no concurrent agents)

```python
# rigg/config.py

class LiteMode:

    def __init__(self):
        self.db_path = Path.home() / ".rigg" / "rigg.db"

    async def get_db(self) -> aiosqlite.Connection:
        return await aiosqlite.connect(self.db_path)

    def warn_limitations(self):
        console.print("""
[yellow]Running in lite mode (--no-docker)[/yellow]

Limitations:
- No vector embeddings (semantic search disabled)
- No Redis (blackboard uses local files)
- No sandbox (code runs without isolation)
- Single agent only (no swarm/jury patterns)

For full features, run with Docker.
""")
```

## Resource Management

### Dynamic Limits Based on History

rigg learns from past resource usage to set appropriate limits:

```python
# rigg/resources/limits.py

class DynamicLimits:
    """Calculate limits based on historical usage."""

    async def calculate_max_agents(self) -> int:
        """Dynamic agent limit based on RAM and history."""

        # Get available RAM
        free_mb = psutil.virtual_memory().available // (1024 * 1024)

        # Base calculation
        base_max = min(20, max(1, (free_mb - 2048) // 500))

        # Adjust based on history
        history = await self.db.fetch("""
            SELECT AVG(peak_memory_mb) as avg_peak,
                   MAX(peak_memory_mb) as max_peak
            FROM agent_runs
            WHERE started_at > NOW() - INTERVAL '7 days'
        """)

        if history['avg_peak']:
            # Reduce limit if agents historically use more RAM
            adjustment = history['avg_peak'] / 400  # Expected 400MB/agent
            return int(base_max / adjustment)

        return base_max

    async def get_memory_warning_threshold(self) -> int:
        """MB threshold for memory warning."""

        # Check historical patterns
        incidents = await self.db.fetchval("""
            SELECT COUNT(*) FROM resource_incidents
            WHERE type = 'oom' AND occurred_at > NOW() - INTERVAL '30 days'
        """)

        # Be more conservative if OOMs have occurred
        if incidents > 0:
            return 3072  # Warn at 3GB remaining
        return 2048  # Default: warn at 2GB remaining
```

### Warn Before Exceeding Memory

```python
# rigg/hooks/resource_check.py

async def check_before_agent_spawn() -> ResourceCheckResult:
    """Pre-spawn resource check with warning."""

    limits = DynamicLimits()
    current = await get_current_resources()

    # Calculate projected usage
    avg_agent_memory = await limits.get_avg_agent_memory()
    projected_free = current.free_mb - avg_agent_memory

    if projected_free < limits.warning_threshold:
        return ResourceCheckResult(
            allowed=True,  # Allow but warn
            warning=f"""
⚠️  MEMORY WARNING

Current: {current.free_mb}MB free
Projected after spawn: {projected_free}MB free
Threshold: {limits.warning_threshold}MB

Based on your history, agents use ~{avg_agent_memory}MB average.
You may experience slowdowns or OOM errors.

Continue anyway? (rigg will proceed unless you cancel)
""",
            requires_ack=True,
        )

    return ResourceCheckResult(allowed=True)
```

### Graceful Degradation on Resource Pressure

```python
# rigg/resources/pressure.py

class ResourcePressureHandler:
    """Handle resource pressure gracefully."""

    async def on_high_memory(self):
        """Triggered when memory > 80% used."""

        # 1. Stop accepting new agents
        await self.set_agent_limit(0)

        # 2. Notify running agents to wrap up
        await self.broadcast({
            "type": "resource_pressure",
            "action": "complete_current_task",
            "reason": "High memory usage detected",
        })

        # 3. Log incident
        await self.log_incident("memory_pressure", {
            "free_mb": psutil.virtual_memory().available // (1024 * 1024),
            "active_agents": await self.get_active_count(),
        })

    async def on_critical_memory(self):
        """Triggered when memory > 95% used."""

        # 1. Terminate non-essential agents
        agents = await self.get_running_agents()
        for agent in agents:
            if agent.priority != "critical":
                await agent.terminate(reason="Critical memory pressure")

        # 2. Clear Redis caches
        await self.redis.flushdb()

        # 3. Force garbage collection
        import gc
        gc.collect()
```

### Priority Resolution

When resolving credentials, rigg checks in order:

```python
# rigg/auth/resolver.py

def get_api_key(service: str) -> str | None:
    """Resolve API key with priority chain."""

    # 1. CLI argument (highest priority)
    if cli_args.get(f"{service}_api_key"):
        return cli_args[f"{service}_api_key"]

    # 2. Stored in auth.json
    auth = load_auth()
    if service in auth.get("api_keys", {}):
        return auth["api_keys"][service]

    # 3. OAuth token (if applicable)
    if service in auth.get("oauth_tokens", {}):
        token = auth["oauth_tokens"][service]
        if not is_expired(token):
            return token["access_token"]
        # Try refresh
        refreshed = await refresh_oauth_token(service)
        if refreshed:
            return refreshed["access_token"]

    # 4. Environment variable (lowest priority)
    env_var = f"{service.upper()}_API_KEY"
    if os.environ.get(env_var):
        return os.environ[env_var]

    return None
```

### Configuration API

```python
# rigg/config.py

from pydantic import BaseModel
from typing import Optional

class TimeoutConfig(BaseModel):
    default: int = 30000
    tools: dict[str, int] = {}

    def get_timeout(self, tool: str) -> int:
        return self.tools.get(tool, self.default)

class EditorConfig(BaseModel):
    keybindings: str = "vim"  # vim, emacs, nano, custom
    custom_bindings: dict[str, str] = {}

    def get_bindings(self) -> dict[str, str]:
        presets = {
            "vim": {"j": "down", "k": "up", "h": "left", "l": "right"},
            "emacs": {"ctrl+n": "down", "ctrl+p": "up", "ctrl+b": "left", "ctrl+f": "right"},
            "nano": {"ctrl+x": "exit", "ctrl+o": "save"},
        }
        base = presets.get(self.keybindings, {})
        return {**base, **self.custom_bindings}

class ThemeConfig(BaseModel):
    mode: str = "auto"  # auto, light, dark, custom
    custom: Optional[dict[str, str]] = None

    def resolve(self) -> dict[str, str]:
        if self.mode == "auto":
            # Detect from terminal
            return detect_terminal_theme()
        elif self.mode in ("light", "dark"):
            return BUILTIN_THEMES[self.mode]
        elif self.mode == "custom" and self.custom:
            return self.custom
        return BUILTIN_THEMES["dark"]

class Config(BaseModel):
    database: dict
    redis: dict
    timeouts: TimeoutConfig
    editor: EditorConfig
    theme: ThemeConfig
    resources: dict
    pricing: dict

def load_config() -> Config:
    """Load config with JSONC parsing."""
    config_path = Path.home() / ".rigg" / "config.json"

    if not config_path.exists():
        return Config(**DEFAULT_CONFIG)

    # Parse JSONC (strip comments)
    text = config_path.read_text()
    text = re.sub(r'//.*$', '', text, flags=re.MULTILINE)
    text = re.sub(r'/\*.*?\*/', '', text, flags=re.DOTALL)

    data = json.loads(text)
    return Config(**data)
```

## Pricing

### Live Price Fetching

rigg fetches current pricing weekly with embedded fallback:

```python
# rigg/pricing/fetcher.py

PRICING_URL = "https://api.anthropic.com/v1/pricing"
FALLBACK_PRICING = {
    "claude-opus-4-5-20251101": {"input": 15.0, "output": 75.0},  # per 1M tokens
    "claude-sonnet-4-20250514": {"input": 3.0, "output": 15.0},
    "claude-haiku-3-5-20241022": {"input": 0.25, "output": 1.25},
}
CACHE_DURATION_DAYS = 7

async def get_pricing() -> dict:
    """Get current pricing, fetch if stale."""

    cache = load_pricing_cache()

    if cache and not is_stale(cache, days=CACHE_DURATION_DAYS):
        return cache["pricing"]

    # Try to fetch live
    try:
        response = await httpx.get(
            PRICING_URL,
            timeout=10,
            headers={"x-api-key": get_api_key("anthropic")}
        )
        if response.status_code == 200:
            pricing = response.json()
            save_pricing_cache(pricing)
            return pricing
    except Exception as e:
        console.print(f"[yellow]Could not fetch live pricing: {e}[/yellow]")

    # Fallback to embedded
    console.print("[yellow]Using embedded fallback pricing (may be outdated)[/yellow]")
    return FALLBACK_PRICING
```

### Cost Tracking

Track costs at multiple granularities:

```python
# rigg/pricing/tracker.py

class CostTracker:
    """Track API costs per turn, session, and aggregate."""

    def __init__(self, db, pricing):
        self.db = db
        self.pricing = pricing

    async def record_usage(
        self,
        session_id: str,
        model: str,
        input_tokens: int,
        output_tokens: int,
    ):
        """Record token usage and calculate cost."""

        model_pricing = self.pricing.get(model, FALLBACK_PRICING.get(model))
        if not model_pricing:
            return

        cost = (
            (input_tokens / 1_000_000) * model_pricing["input"] +
            (output_tokens / 1_000_000) * model_pricing["output"]
        )

        await self.db.execute("""
            INSERT INTO usage (session_id, model, input_tokens, output_tokens, cost_usd, timestamp)
            VALUES ($1, $2, $3, $4, $5, NOW())
        """, session_id, model, input_tokens, output_tokens, cost)

        # Check alerts
        await self.check_alerts(session_id)

    async def get_session_total(self, session_id: str) -> float:
        """Get total cost for a session."""
        return await self.db.fetchval("""
            SELECT COALESCE(SUM(cost_usd), 0)
            FROM usage WHERE session_id = $1
        """, session_id)

    async def get_daily_total(self) -> float:
        """Get today's total cost."""
        return await self.db.fetchval("""
            SELECT COALESCE(SUM(cost_usd), 0)
            FROM usage WHERE timestamp > CURRENT_DATE
        """)

    async def check_alerts(self, session_id: str):
        """Check and emit budget alerts."""

        config = load_config()
        alerts = config.pricing.get("alerts", {})
        budget = config.pricing.get("budget", {})

        session_total = await self.get_session_total(session_id)
        daily_total = await self.get_daily_total()

        # Session threshold alert
        if alerts.get("session_threshold_usd"):
            if session_total >= alerts["session_threshold_usd"]:
                await self.emit_alert(
                    "session_threshold",
                    f"Session cost: ${session_total:.2f} (threshold: ${alerts['session_threshold_usd']})"
                )

        # Daily threshold alert
        if alerts.get("daily_threshold_usd"):
            if daily_total >= alerts["daily_threshold_usd"]:
                await self.emit_alert(
                    "daily_threshold",
                    f"Daily cost: ${daily_total:.2f} (threshold: ${alerts['daily_threshold_usd']})"
                )

        # Hard limits
        if budget.get("session_limit_usd"):
            if session_total >= budget["session_limit_usd"]:
                raise BudgetExceededError(f"Session budget exceeded: ${session_total:.2f}")

        if budget.get("daily_limit_usd"):
            if daily_total >= budget["daily_limit_usd"]:
                raise BudgetExceededError(f"Daily budget exceeded: ${daily_total:.2f}")

    async def emit_alert(self, alert_type: str, message: str):
        """Emit budget alert to user."""

        # Only emit once per session per alert type
        already_alerted = await self.db.fetchval("""
            SELECT 1 FROM alerts WHERE session_id = $1 AND type = $2
        """, self.session_id, alert_type)

        if not already_alerted:
            console.print(f"[yellow]BUDGET ALERT: {message}[/yellow]")
            await self.db.execute("""
                INSERT INTO alerts (session_id, type, message, timestamp)
                VALUES ($1, $2, $3, NOW())
            """, self.session_id, alert_type, message)
```

### Budget Thresholds with Notifications

```python
# rigg/pricing/notifications.py

async def setup_budget_notifications():
    """Configure budget notifications."""

    config = load_config()

    if not config.pricing.get("budget"):
        return

    # Register hook for cost tracking
    register_hook("post_api_call", track_and_notify)

async def track_and_notify(event: dict):
    """PostToolUse hook for cost tracking."""

    if event.get("type") != "api_response":
        return

    usage = event.get("usage", {})
    tracker = get_cost_tracker()

    try:
        await tracker.record_usage(
            session_id=event["session_id"],
            model=event["model"],
            input_tokens=usage.get("input_tokens", 0),
            output_tokens=usage.get("output_tokens", 0),
        )
    except BudgetExceededError as e:
        # Hard limit - block further usage
        return {
            "result": "block",
            "reason": str(e),
        }
```
